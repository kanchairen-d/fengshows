export default {
 async fetch(request, env, ctx) {
 const url = new URL(request.url);
 const path = url.pathname.split('/').filter(Boolean);

 // 根路径 → 显示 Web 页面
 if (path.length === 0) {
     const domain = url.hostname + (url.port ? ':' + url.port : '');
     const hasAccount = !!(env.PHONE && env.PASSWORD);
     return new Response(buildPage(domain, hasAccount), {
         headers: { "Content-Type": "text/html; charset=utf-8" }
     });
 }

 const id = path[0] || 'fhzw';
 const chid = CHANNELS[id];
 if (!chid) {
 return new Response("Invalid Channel ID", { status: 400 });
 }
 let token = null;
 let quality = 'hd';
 if (env.PHONE && env.PASSWORD) {
 token = await getValidToken(env);
 if (token) {
 quality = 'fhd';
 }
 }
 const apiUrl = `https://api.fengshows.cn/hub/live/auth-url?live_qa=${quality}&live_id=${chid}`;
 const headers = {
 "User-Agent": "Mozilla/5.0 (Linux; Android 10; SM-G960U) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/88.0.4324.181 Mobile Safari/537.36",
 "Referer": "https://www.fengshows.com/live",
 "Origin": "https://www.fengshows.com",
 };
 if (token) {
 headers["Token"] = token;
 }
 try {
 const response = await fetch(apiUrl, { headers });
 const data = await response.json();
 if (data && data.data && data.data.live_url) {
 return Response.redirect(data.data.live_url, 302);
 } else {
 if (quality === 'fhd') {
 const fallbackUrl = `https://api.fengshows.cn/hub/live/auth-url?live_qa=hd&live_id=${chid}`;
 const fallbackRes = await fetch(fallbackUrl, { headers: { ...headers, Token: '' } });
 const fallbackData = await fallbackRes.json();
 if (fallbackData && fallbackData.data && fallbackData.data.live_url) {
 return Response.redirect(fallbackData.data.live_url, 302);
 }
 }
 return new Response(JSON.stringify(data), { status: 500, headers: { "Content-Type": "application/json" } });
 }
 } catch (e) {
 return new Response(e.message, { status: 500 });
 }
 }
};
async function getValidToken(env) {
 if (cachedToken && validateJWT(cachedToken)) {
 return cachedToken;
 }
 const loginUrl = "https://m.fengshows.com/api/v3/mp/user/login";
 const body = {
 "code": "86",
 "keep_alive": false,
 "password": env.PASSWORD,
 "phone": env.PHONE,
 };
 try {
 const response = await fetch(loginUrl, {
 method: "POST",
 headers: { "Content-Type": "application/json" },
 body: JSON.stringify(body)
 });
 const data = await response.json();
 if (data.message === "ok") {
 cachedToken = data.data.token;
 return cachedToken;
 }
 } catch (e) {
 console.error("Login failed:", e);
 }
 return null;
}
function validateJWT(token) {
 try {
 const parts = token.split('.');
 if (parts.length !== 3) return false;
 let payloadBase64 = parts[1].replace(/-/g, '+').replace(/_/g, '/');
 while (payloadBase64.length % 4) {
 payloadBase64 += '=';
 }
 const payload = JSON.parse(atob(payloadBase64));
 const exp = payload.exp;
 if (!exp) return true;
 return exp > (Math.floor(Date.now() / 1000) + 300);
 } catch (e) {
 return false;
 }
}
